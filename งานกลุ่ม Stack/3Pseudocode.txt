Class Command:
    type: String          // "INSERT", "DELETE", "REPLACE"
    index: Integer
    data: String          // ข้อความที่แทรก หรือ ข้อความที่ถูกลบ
    replacedText: String  // บันทึกข้อความเดิมก่อนถูกทับ (ใช้เฉพาะ REPLACE)

Class DeltaTextEditor:
    text: String = ""
    undoStack: Stack of Command
    redoStack: Stack of Command

    Function executeCommand(cmd):
        IF cmd.type == "INSERT":
            text.insert(cmd.index, cmd.data)
        ELSE IF cmd.type == "DELETE":
            cmd.data = text.substring(cmd.index, cmd.index + cmd.length) // เผื่อ Undo
            text.delete(cmd.index, cmd.length)
        ELSE IF cmd.type == "REPLACE":
            cmd.replacedText = text.substring(cmd.index, cmd.index + cmd.data.length)
            text.replace(cmd.index, cmd.data)
        
        undoStack.push(cmd)
        redoStack.clear() // เคลียร์ Redo Stack เมื่อมีการแก้ไขใหม่

    Function undo():
        IF undoStack.isEmpty(): RETURN
        cmd = undoStack.pop()
        
        // ประมวลผล Inverse Operation
        IF cmd.type == "INSERT":
            text.delete(cmd.index, cmd.data.length)
        ELSE IF cmd.type == "DELETE":
            text.insert(cmd.index, cmd.data)
        ELSE IF cmd.type == "REPLACE":
            text.replace(cmd.index, cmd.replacedText)
            
        redoStack.push(cmd)

    Function redo():
        IF redoStack.isEmpty(): RETURN
        cmd = redoStack.pop()
        
        // ประมวลผลคำสั่งเดิมซ้ำ
        IF cmd.type == "INSERT":
            text.insert(cmd.index, cmd.data)
        ELSE IF cmd.type == "DELETE":
            text.delete(cmd.index, cmd.data.length)
        ELSE IF cmd.type == "REPLACE":
            text.replace(cmd.index, cmd.data)
            
        undoStack.push(cmd)
